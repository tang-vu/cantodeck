cmake_minimum_required(VERSION 3.22)
project(CantoDeck VERSION 0.1.0 LANGUAGES C CXX)
set(CANTO_REVISION "source-archive")
if(EXISTS "${CMAKE_SOURCE_DIR}/.git")
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
 OUTPUT_VARIABLE CANTO_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE CANTO_GIT_RESULT ERROR_QUIET)
if(CANTO_GIT_RESULT EQUAL 0)
 execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
  OUTPUT_VARIABLE CANTO_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
 if(NOT CANTO_DIRTY STREQUAL "")
  string(APPEND CANTO_REVISION "-dirty")
 endif()
else()
 set(CANTO_REVISION "source-archive")
endif()
endif()
string(TIMESTAMP CANTO_CONFIGURED_UTC "%Y-%m-%dT%H:%M:%SZ" UTC)
configure_file(app/BuildInfo.h.in generated/BuildInfo.h @ONLY)
set(CMAKE_CXX_STANDARD 20)
include(FetchContent)
if(EXISTS "${CMAKE_SOURCE_DIR}/vendor/JUCE/CMakeLists.txt")
  add_subdirectory(vendor/JUCE)
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 91ad83ae34a81e0833b1a2b0866f54846370ae53 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_gui_app(CantoDeck PRODUCT_NAME "CantoDeck" COMPANY_NAME "CantoDeck" VERSION 0.1.0)
target_sources(CantoDeck PRIVATE app/Main.cpp engine/AudioEngine.cpp)
if(WIN32)
 target_sources(CantoDeck PRIVATE platform/windows/NativeWasapi.cpp)
 target_link_libraries(CantoDeck PRIVATE avrt ole32 uuid)
endif()
target_include_directories(CantoDeck PRIVATE ${CMAKE_SOURCE_DIR} "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(CantoDeck PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_ASIO=0 JUCE_WASAPI=1 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 JUCE_USE_MP3AUDIOFORMAT=0)
target_link_libraries(CantoDeck PRIVATE juce::juce_audio_utils juce::juce_dsp)
if(MSVC)
 target_compile_options(CantoDeck PRIVATE /utf-8)
endif()
enable_testing()
add_executable(core_tests tests/CoreTests.cpp)
target_include_directories(core_tests PRIVATE ${CMAKE_SOURCE_DIR})
add_test(NAME core_tests COMMAND core_tests)
set_tests_properties(core_tests PROPERTIES TIMEOUT 60)
add_executable(clock_scheduling_tests tests/ClockSchedulingTests.cpp)
target_include_directories(clock_scheduling_tests PRIVATE ${CMAKE_SOURCE_DIR})
add_test(NAME clock_scheduling COMMAND clock_scheduling_tests)
add_test(NAME clock_scheduling_no_jitter COMMAND clock_scheduling_tests --no-jitter)
set_tests_properties(clock_scheduling clock_scheduling_no_jitter PROPERTIES TIMEOUT 60)
add_executable(stream_tests tests/WavStreamTests.cpp)
target_include_directories(stream_tests PRIVATE ${CMAKE_SOURCE_DIR})
target_compile_definitions(stream_tests PRIVATE JUCE_USE_CURL=0 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 JUCE_USE_MP3AUDIOFORMAT=0)
target_link_libraries(stream_tests PRIVATE juce::juce_audio_formats)
add_test(NAME wav_streaming COMMAND stream_tests)
set_tests_properties(wav_streaming PROPERTIES TIMEOUT 30)
if(WIN32)
 add_executable(native_backend_tests tests/NativeBackendTests.cpp platform/windows/NativeWasapi.cpp)
 target_include_directories(native_backend_tests PRIVATE ${CMAKE_SOURCE_DIR})
 target_link_libraries(native_backend_tests PRIVATE avrt ole32 uuid)
 add_test(NAME native_backend_validation COMMAND native_backend_tests)
 set_tests_properties(native_backend_validation PROPERTIES TIMEOUT 15)
endif()
install(TARGETS CantoDeck RUNTIME DESTINATION .)
set(CPACK_GENERATOR ZIP)
include(CPack)
